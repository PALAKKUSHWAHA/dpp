int main() {
    long long a, b;
    cin >> a >> b; // Read the input values for a and b

    vector<long long> sequence; // Vector to store the transformation sequence
    long long current = b; // Start from b and work backwards

    // Loop until we reach a or current is less than a
    while (current >= a) {
        sequence.push_back(current); // Add current to the sequence
        
        // Check if current is even and can be divided by 2
        if (current % 2 == 0) {
            current /= 2; // Reverse the multiplication by 2
        } 
        // Check if current ends with 1, meaning it could have been formed by appending 1
        else if (current % 10 == 1) {
            current = (current - 1) / 10; // Reverse the append operation
        } 
        // If neither condition is met, break out of the loop
        else {
            break;
        }
    }

    // Check if we have reached a
    if (current == a) {
        cout << "YES" << endl; // Output YES
        cout << sequence.size() << endl; // Output the length of the sequence
        
        // Since we've built the sequence backwards, we need to reverse it for output
        reverse(sequence.begin(), sequence.end());
        
        // Output the transformation sequence
        for (long long num : sequence) {
            cout << num << " ";
        }
        cout << endl;
    } else {
        cout << "NO" << endl; // Output NO if transformation is not possible
    }

    return 0;
}

https://leetcode.com/problems/permutations/description/

class Solution {

// Create an answer vector
// base conditions -> when index is at end -> terminate the recursion/return
//when I am at index i -> provide it with all the choices from i to size()-1
//use swap -> every ith index to every number from i to n-1 (size -1)
// I move forward to i+1
// Basically fix index i with diff choices

public:
    vector<vector<int>> ans;
    void generate(vector<int> &nums, int index){

        if(index == nums.size()){
            ans.push_back(nums);
            return;
        }
        for(int i = index;i<nums.size();i++){
            swap(nums[index], nums[i]);
            generate(nums,index+1);
            swap(nums[index],nums[i]);
        }
        return;
    }
    vector<vector<int>> permute(vector<int>& nums) {
        generate(nums,0);
        return ans;
    }
};


https://leetcode.com/problems/generate-parentheses/
class Solution {
// maintain number of opening and closing brackets
// base condition 1 -> if both are 0 -> reached end with correct sequence -> push it answer
// base condition 2 -> when number of closing brackets(to put) > opening brackets -> return because false sequence
// if opening bracket count != 0 put opening bracket in cs and move forward
// if closing bracket count != 0 put closing bracket in cs and move forward

public:
    vector<string> ans;
    void generate(int opening, int closing, string &cs){
        if(opening == 0 && closing == 0){
            ans.push_back(cs);
            return;
        }
        if(closing < opening){
            return; // this is the false state or incorrect state
        }
        if(opening > 0){
            cs+='(';
            generate(opening -1, closing, cs);
            cs.pop_back();
        }
        if(closing > 0){
            cs+=')';
            generate(opening , closing-1, cs);
            cs.pop_back();
        }
        return;
    }
    vector<string> generateParenthesis(int n) {
        string cs = "";
        generate(n,n,cs);
        return ans;
    }
};

class Solution {
// s -> 0 and e -> n-1
// m is found
// if nums[m] < nums[end] -> e = m
// else -> s = m+1 
// return s or e
// pivot is returned from the above function
// new start and end -> for simple binary search
// target <= nums[end] -> s = pivot
// else -> end = pivot - 1
// apply simple binary search return ans 
// return m if found
// return -1 out of loop 
public:
    int findMin(vector<int>& nums) {
        int s = 0, e = nums.size()-1;
        while(s < e){
            int m = s + (e-s)/2;
            if(nums[m] < nums[e]){
                e = m;
            }else{
                s = m+1;
            }
        }
        return s;
    }
    int search(vector<int>& nums, int target) {
        int pivot = findMin(nums);
        int s = 0, e =  nums.size() - 1;
        if(target <= nums[e]){
            s = pivot;
        }else{
            e = pivot - 1;
        }
        while(s <= e){
            int m = s + (e-s)/2;
            if(nums[m] == target){
                return m;
            }
            if(nums[m] > target){
                e = m-1;
            }else{
                s = m+1;
            }
        }
        return -1;
    }
};




class Solution {
// s = 0 e = n-1
// define mid;
// if mid == target -> return mid;
// if mid < nums[e] (left array) -> target is between mid -> e
// if between mid -> e ->s = mid+1
// else e = mid - 1;
// if mid > nums[e] (right array) -> target is between s and mid;
// if between s and mid -> e = mid - 1;
// else s = mid+1
public:
    int search(vector<int>& nums, int target) {
        int s = 0, e = nums.size() - 1;
        while(s <= e){
            int mid = s + (e-s)/2;
            if(nums[mid] == target){
                return mid;
            }
            if(nums[mid] < nums[e]){
                if(target >= nums[mid] && target <= nums[e]){
                    s = mid+1;
                }else{
                    e = mid -1;
                }
            }else{
                if(target >= nums[s] && target <= nums[mid]){
                    e = mid -1;
                }else{
                    s = mid+1;
                }
            }
        }
        return -1;
    }
};

/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
//BFS
// make a queue
// push root with 1 height
// while !q.empty -> push child with currheight + 1; 
// print maxheight

class Solution {
public:
    int maxDepth(TreeNode* root) {
        int maxHeight = 1;
        queue <pair<TreeNode*,int>> q; // node and it's curr height
        if(!root) return 0;
        q.push({root,1});
        while(!q.empty()){
            maxHeight = max (maxHeight, q.front().second);
            TreeNode* node = q.front().first;
            int height = q.front().second;
            q.pop();
            if(node -> left) q.push({node -> left, height + 1});
            if(node -> right) q.push({node -> right, height + 1});
        }
        return maxHeight;
    }
};




/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */


// DFS
// maxHeight variable global
// if node -> null -> return;
// check if currHeight > maxHeight -> update maxHeight
// if(child) -> go to child with height + 1;
class Solution {
public:
    int maxHeight = 0;

    void dfs(TreeNode * node, int currHeight){
        if(!node) return;
        maxHeight = max (maxHeight, currHeight);
        if(node -> left) dfs(node -> left, currHeight + 1);
        if(node -> right) dfs(node -> right, currHeight + 1);
    }

    int maxDepth(TreeNode* root) {
        dfs(root,1);
        return maxHeight;
    }
};

// first make 2 stacks -> one to generate output stack(stack1), one to give output(stack2)
// push root in stack1
// loop till stack1 is empty
// push topmost element of stack1 in stack2
// pop stack1 elemnt
// push popped elemt left and then right in stack 1
// pop till stack2 is empty and return ans;
class Solution {
public:
    vector<int> postorderTraversal(TreeNode* root) {
        stack<TreeNode*> stack1,stack2;
        vector<int> ans;
        if (!root) return ans;
        stack1.push(root);
        while(!stack1.empty()){
            stack2.push(stack1.top());
            TreeNode* node = stack1.top();
            stack1.pop();
            if(node -> left) stack1.push(node -> left);
            if(node -> right) stack1.push(node -> right);
        }

        while(!stack2.empty()){
            ans.push_back(stack2.top() -> val);
            stack2.pop();
        }
        return ans;
        

    }
